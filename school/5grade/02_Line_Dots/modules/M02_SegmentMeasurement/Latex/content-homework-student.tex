  {Домашняя практика}
  {\modulemark{M02-H} Измеряем и строим отрезки}

\studentnameblock

\begin{practicebox}[1. Длина по показаниям]
  \begin{enumerate}[label=\alph*)]
    \item начало $0$, конец $6\text{ см }2\text{ мм}$:
      \answerblank[45mm];
    \item начало $1\text{ см }7\text{ мм}$, конец
      $7\text{ см }5\text{ мм}$: \answerblank[55mm];
    \item начало $3\text{ см }4\text{ мм}$, конец
      $9\text{ см}$: \answerblank[55mm].
  \end{enumerate}
\end{practicebox}

\begin{practicebox}[2. Построения]
  Постройте $AB=4\text{ см }3\text{ мм}$ и
  $CD=6\text{ см }1\text{ мм}$.
  \workarea{50mm}
\end{practicebox}

\begin{practicebox}[3. Равные отрезки]
  $MN=52\text{ мм}$, $PK=5\text{ см }2\text{ мм}$. Равны ли отрезки?
  Объясните. \writinglines{1}
\end{practicebox}

\begin{warningbox}[4. Исправьте ошибку]
  «От отметки $2\text{ см}$ до отметки $8\text{ см }4\text{ мм}$ длина
  равна $8\text{ см }4\text{ мм}$».
  \writinglines{2}
\end{warningbox}
