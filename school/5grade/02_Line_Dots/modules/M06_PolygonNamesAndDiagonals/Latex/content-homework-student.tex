\worksheettitle
  {Домашняя практика}
  {\modulemark{M06-H} Названия и диагонали}

\studentnameblock

\begin{practicebox}[1. Проверка названия]
  Для пятиугольника $ABCDE$ отметьте правильные имена:
  \[
    \square\ ABCDE\quad
    \square\ CDEAB\quad
    \square\ AEDCB\quad
    \square\ ACBED.
  \]
  Объясните ошибку в неверной записи. \writinglines{2}
\end{practicebox}

\begin{practicebox}[2. Стороны и диагонали]
  В пятиугольнике $ABCDE$ определите:
  \[
    AB\text{ --- }\answerblank[22mm],\quad
    AD\text{ --- }\answerblank[22mm],\quad
    EA\text{ --- }\answerblank[22mm].
  \]
\end{practicebox}

\begin{practicebox}[3. Из одной вершины]
  Запишите все диагонали пятиугольника $KLMNP$, выходящие из $K$.
  \answerblank[80mm]
\end{practicebox}

\begin{practicebox}[4. Разбиение]
  Диагональ $BD$ разделила пятиугольник $ABCDE$ на две части.
  Назовите обе части по их границам. \writinglines{2}
\end{practicebox}

\begin{warningbox}[5. Исправьте ошибку]
  «$AB$ и $BA$ --- разные стороны, а $AC$ является стороной, потому что
  обе точки --- вершины». Назовите две причины и запишите исправление.
  \writinglines{3}
\end{warningbox}
