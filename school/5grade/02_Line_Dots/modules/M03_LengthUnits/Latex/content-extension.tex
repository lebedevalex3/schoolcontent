\worksheettitle
  {M03\(\star\). Выбираем удобную единицу}
  {\modulemark{M03\(\star\)} Усложнение}

\studentnameblock

\begin{notebox}[Главный вопрос]
  Одну длину можно записать по-разному. Удобная единица делает сравнение и
  вычисление короче, но выбор нужно объяснить.
\end{notebox}

\begin{practicebox}[1. Сравните двумя способами]
  Сравните $2\text{ м }45\text{ см}$ и $238\text{ см}$:
  \begin{enumerate}[label=\alph*)]
    \item переведите обе длины в сантиметры;
    \item разложите $238$ см на метры и сантиметры.
  \end{enumerate}
  Какой способ оказался удобнее? \writinglines{2}
\end{practicebox}

\begin{practicebox}[2. Выберите единицу]
  Найдите сумму $3\text{ м }8\text{ см}+95\text{ см}$. Решите в выбранной
  вами единице, затем запишите ответ в метрах и сантиметрах.
  \writinglines{2}
\end{practicebox}

\begin{practicebox}[3. Найдите все записи]
  Запишите $2500$ мм:
  \begin{enumerate}[label=\alph*)]
    \item только в сантиметрах;
    \item в метрах и сантиметрах;
    \item в метрах, сантиметрах и миллиметрах.
  \end{enumerate}
\end{practicebox}

\begin{checkpointbox}[Исследование]
  Придумайте две по-разному записанные, но равные длины. Объясните равенство
  двумя способами. \writinglines{3}
\end{checkpointbox}
